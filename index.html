/** * AGENCY COMMAND CENTER — BACKEND (Google Apps Script) * ----------------------------------------------------- * Lee las pestañas Clients, Tasks, Content, Campaigns, Approvals * de un Google Sheet y las devuelve como JSON via una URL Web App. * * SETUP: * 1. Abre tu Google Sheet (MSMP_Client_Master convertido a Google Sheets) * 2. Extensiones > Apps Script * 3. Borra el contenido de Code.gs y pega todo este archivo * 4. Reemplaza SHEET_ID abajo con el ID real de tu sheet * (está en la URL: docs.google.com/spreadsheets/d/AQUÍ_VA_EL_ID/edit) * 5. Deploy > New deployment > tipo "Web app" * - Execute as: Me * - Who has access: Anyone * 6. Copia la URL que te da (termina en /exec) — esa va en el HTML * 7. Cada vez que edites este código, tienes que hacer un NUEVO deployment * (o "Manage deployments" > editar > nueva versión) para que se actualice * * ESCRITURA (doPost): * - Actualizar una celda: { sheetName, idColumn, idValue, field, value } * - Agregar una fila nueva: { action:'insert', sheetName, idColumn, idPrefix, values:{...} } * El ID nuevo se genera solo (ej. CON-017) según el máximo existente + 1. */ // ==== CONFIG — EDITA ESTO ==== const SHEET_ID = '1LBFwOE71vWZasEtcKS655yfLf1pT7s0e8yZrN0qBm7U'; const TABS = { clients: 'Clients', tasks: 'Tasks', content: 'Content', campaigns: 'Campaigns', approvals: 'Approvals' }; // ==== ENDPOINT ==== function doGet(e) { try { const ss = SpreadsheetApp.openById(SHEET_ID); const data = {}; for (const key in TABS) { data[key] = sheetToObjects(ss.getSheetByName(TABS[key])); } return jsonResponse({ ok: true, updated: new Date().toISOString(), data: data }); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); if (payload.action === 'insert') { return handleInsert(ss, payload); } if (payload.action === 'delete') { return handleDelete(ss, payload); } return handleUpdate(ss, payload); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } } function handleDelete(ss, payload) { const sheetName = payload.sheetName; const idColumn = payload.idColumn; const idValue = payload.idValue; const sheet = ss.getSheetByName(sheetName); if (!sheet) throw new Error('Pestaña no encontrada: ' + sheetName); const data = sheet.getDataRange().getValues(); const headers = data[0].map(function (h) { return String(h).trim(); }); const idColIdx = headers.indexOf(idColumn); if (idColIdx === -1) throw new Error('Columna ID no encontrada: ' + idColumn); for (let i = 1; i < data.length; i++) { if (String(data[i][idColIdx]) === String(idValue)) { sheet.deleteRow(i + 1); return jsonResponse({ ok: true, action: 'delete', sheetName: sheetName, idValue: idValue }); } } throw new Error('Fila no encontrada con ' + idColumn + ' = ' + idValue); } function handleUpdate(ss, payload) { const sheetName = payload.sheetName; const idColumn = payload.idColumn; const idValue = payload.idValue; const field = payload.field; const value = payload.value; const sheet = ss.getSheetByName(sheetName); if (!sheet) throw new Error('Pestaña no encontrada: ' + sheetName); const data = sheet.getDataRange().getValues(); const headers = data[0].map(function (h) { return String(h).trim(); }); const idColIdx = headers.indexOf(idColumn); const fieldColIdx = headers.indexOf(field); if (idColIdx === -1) throw new Error('Columna ID no encontrada: ' + idColumn); if (fieldColIdx === -1) throw new Error('Columna no encontrada: ' + field); for (let i = 1; i < data.length; i++) { if (String(data[i][idColIdx]) === String(idValue)) { sheet.getRange(i + 1, fieldColIdx + 1).setValue(value); return jsonResponse({ ok: true, action: 'update', sheetName: sheetName, idValue: idValue, field: field, value: value }); } } throw new Error('Fila no encontrada con ' + idColumn + ' = ' + idValue); } function handleInsert(ss, payload) { const sheetName = payload.sheetName; const idColumn = payload.idColumn; const idPrefix = payload.idPrefix; const values = payload.values || {}; const sheet = ss.getSheetByName(sheetName); if (!sheet) throw new Error('Pestaña no encontrada: ' + sheetName); const data = sheet.getDataRange().getValues(); const headers = data[0].map(function (h) { return String(h).trim(); }); const idColIdx = headers.indexOf(idColumn); if (idColIdx === -1) throw new Error('Columna ID no encontrada: ' + idColumn); // generar el siguiente ID secuencial tipo PREFIX-001 let maxNum = 0; for (let i = 1; i < data.length; i++) { const idVal = String(data[i][idColIdx] || ''); const match = idVal.match(/(\d+)\s*$/); if (match) maxNum = Math.max(maxNum, parseInt(match[1], 10)); } const nextId = idPrefix + '-' + String(maxNum + 1).padStart(3, '0'); const newRow = headers.map(function (h) { if (h === idColumn) return nextId; return values[h] !== undefined ? values[h] : ''; }); sheet.appendRow(newRow); return jsonResponse({ ok: true, action: 'insert', sheetName: sheetName, newId: nextId, row: newRow }); } // ==== HELPERS ==== // Convierte una pestaña (con encabezados en la fila 1) en un array de objetos function sheetToObjects(sheet) { if (!sheet) return []; const values = sheet.getDataRange().getValues(); if (values.length < 2) return []; const headers = values[0].map(function (h) { return String(h).trim(); }); const rows = values.slice(1); return rows .filter(function (row) { return row.some(function (cell) { return cell !== '' && cell !== null; }); }) .map(function (row) { const obj = {}; headers.forEach(function (h, i) { let val = row[i]; if (val instanceof Date) { val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } obj[h] = val; }); return obj; }); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
